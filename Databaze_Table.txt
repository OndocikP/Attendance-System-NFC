-- ==========================================
-- ZAMESTNANCI
-- ==========================================

create table public.employees (
    id uuid primary key default gen_random_uuid(),

    first_name text not null,
    last_name text not null,

    email text unique,
    phone text,
    position text,

    is_active boolean not null default true,

    created_at timestamptz not null default now()
);


-- ==========================================
-- NFC KARTY
-- ==========================================

create table public.nfc_cards (
    id uuid primary key default gen_random_uuid(),

    card_uid text not null unique,

    -- Karta môže byť nepriradená
    employee_id uuid
        references public.employees(id)
        on delete set null,

    is_active boolean not null default true,

    issued_at timestamptz not null default now(),
    revoked_at timestamptz
);


-- ==========================================
-- DOCHÁDZKA
-- ==========================================

create table public.attendance_records (
    id uuid primary key default gen_random_uuid(),

    employee_id uuid not null
        references public.employees(id)
        on delete restrict,

    card_id uuid
        references public.nfc_cards(id)
        on delete restrict,

    arrival_at timestamptz not null default now(),

    departure_at timestamptz,

    paid boolean not null default false,

    created_at timestamptz not null default now()
);


-- ==========================================
-- INDEXY
-- ==========================================

create index idx_nfc_cards_employee
on public.nfc_cards(employee_id);

create index idx_attendance_employee
on public.attendance_records(employee_id);

create index idx_attendance_recorded_at
on public.attendance_records(recorded_at);

-- ==========================================
-- system_logs
-- ==========================================

create table public.system_logs (
    id bigint generated always as identity primary key,

    created_at timestamptz not null default now(),

    employee_id uuid
        references public.employees(id)
        on delete restrict,

    card_id uuid
        references public.nfc_cards(id)
        on delete restrict,

    msg text
);
